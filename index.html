系統設定
串接步驟
1

建立 Google Sheets

新增試算表，命名「我適文創假勤記錄」。複製網址中 /d/ 和 /edit 之間的 ID，填入上方設定。

2

建立 Apps Script

試算表 → 擴充功能 → Apps Script，刪除預設程式碼，貼入以下程式碼並修改前三行常數後儲存：

const HR_EMAIL = 'hr@company.com'; // 人資信箱 const SHEET_ID = '你的-Sheet-ID'; // Google Sheet ID const COMPANY = '我適文創有限公司'; function doPost(e) { const d = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.openById(SHEET_ID); let sh = ss.getSheetByName('申請記錄') || ss.insertSheet('申請記錄'); if (sh.getLastRow() === 0) sh.appendRow(['編號','申請人','Email','部門','主管','類型', '假別/地點','日期','結束日期','出差時間','事由','備注','送出時間','狀態']); const id = 'REQ-' + Date.now(); sh.appendRow([id,d.name,d.email,d.dept,d.manager,d.type, d.sub,d.startDate,d.endDate,d.tripTime||'', d.reason,d.note,new Date().toLocaleString('zh-TW'),'待審']); // 同步通知人資 + 主管（主管須填 managerEmail 欄位） const subj = `[假勤申請] ${d.name}・${d.type} ${d.startDate}`; const html = buildMail(d, id); MailApp.sendEmail({ to: HR_EMAIL, subject: subj, htmlBody: html }); if (d.managerEmail) MailApp.sendEmail({ to: d.managerEmail, subject: subj, htmlBody: html }); return ContentService .createTextOutput(JSON.stringify({ok:true,id})) .setMimeType(ContentService.MimeType.JSON); } function buildMail(d, id) { const row=(bg,k,v)=> `<tr style="background:${bg}"><td style="padding:9px 12px;color:#888;width:90px">${k}</td> <td style="padding:9px 12px">${v}</td></tr>`; return `<div style="font-family:sans-serif;max-width:520px;color:#1C1B18"> <h2 style="font-size:17px;margin-bottom:14px">📋 假勤申請通知</h2> <table style="width:100%;border-collapse:collapse;font-size:14px"> ${row('#F5F4F0','編號','<b>'+id+'</b>')} ${row('#fff','申請人',d.name+'（'+d.dept+'）')} ${row('#F5F4F0','類型',d.type+(d.sub?'・'+d.sub:''))} ${row('#fff','日期',d.startDate+(d.endDate&&d.endDate!==d.startDate?' ～ '+d.endDate:''))} ${d.tripTime?row('#F5F4F0','出差時間',d.tripTime):''} ${row('#fff','事由',d.reason||'（無）')} </table> <p style="font-size:12px;color:#aaa;margin-top:14px">此信由 ${COMPANY} 假勤系統自動發送</p> </div>`; } function doGet() { const sh=SpreadsheetApp.openById(SHEET_ID).getSheetByName('申請記錄'); if(!sh||sh.getLastRow()<=1) return ContentService.createTextOutput('[]'); const[h,...rows]=sh.getDataRange().getValues(); return ContentService .createTextOutput(JSON.stringify(rows.map(r=>Object.fromEntries(h.map((k,i)=>[k,r[i]]))))) .setMimeType(ContentService.MimeType.JSON); }
3

部署為 Web App

「部署 → 新增部署」→ 類型「網路應用程式」→ 執行身分「我」→ 存取「所有人」→ 按部署 → 複製網址貼入上方設定後儲存。

4

完成！

每筆申請自動寫入 Google Sheets，並同步 email 通知人資與主管。若要加入主管信箱自動帶入，可在申請表單加一欄「主管 Email」，系統會自動寄送。
